cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(image-cmyk LANGUAGES C CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../package.json" COMPONENT_PACKAGE)
string(JSON IMAGE_CMYK_VERSION GET "${COMPONENT_PACKAGE}" version)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IMAGE_DEPENDENCY_PREFIX OR NOT LCMS_SOURCE OR NOT JSON_SOURCE)
  message(FATAL_ERROR "Use scripts/build_cli.py to build verified pinned static dependencies first")
endif()
foreach(path IMAGE_DEPENDENCY_PREFIX LCMS_SOURCE JSON_SOURCE)
  file(TO_CMAKE_PATH "${${path}}" ${path})
endforeach()

if(UNIX)
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()
foreach(library JPEG_LIBRARY PNG_LIBRARY TIFF_LIBRARY ZLIB_LIBRARY)
  unset(${library} CACHE)
endforeach()
find_path(JPEG_INCLUDE_DIR jpeglib.h PATHS "${IMAGE_DEPENDENCY_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(JPEG_LIBRARY NAMES jpeg-static jpeg libjpeg PATHS "${IMAGE_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(PNG_LIBRARY NAMES libpng16_static png16 PATHS "${IMAGE_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(TIFF_LIBRARY NAMES tiff tiff_static PATHS "${IMAGE_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ZLIB_LIBRARY NAMES zlibstatic z PATHS "${IMAGE_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
file(GLOB LCMS_SOURCES CONFIGURE_DEPENDS "${LCMS_SOURCE}/src/*.c")
add_library(lcms2 STATIC ${LCMS_SOURCES})
target_include_directories(lcms2 PUBLIC "${LCMS_SOURCE}/include")
add_library(image_dependencies INTERFACE)
target_include_directories(image_dependencies INTERFACE "${JPEG_INCLUDE_DIR}" "${JSON_SOURCE}/single_include")
target_link_libraries(image_dependencies INTERFACE lcms2 "${JPEG_LIBRARY}" "${TIFF_LIBRARY}" "${PNG_LIBRARY}" "${ZLIB_LIBRARY}")
if(UNIX)
  target_link_libraries(image_dependencies INTERFACE m)
endif()
add_library(image_core STATIC image.cpp)
target_link_libraries(image_core PUBLIC image_dependencies)
add_executable(image-cmyk main.cpp)
target_compile_definitions(image-cmyk PRIVATE IMAGE_CMYK_VERSION="${IMAGE_CMYK_VERSION}")
target_link_libraries(image-cmyk PRIVATE image_core)
if(MSVC)
  target_compile_options(image_core PUBLIC /utf-8)
  target_compile_options(image-cmyk PRIVATE /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(image-cmyk-fixtures fixtures.cpp)
  target_link_libraries(image-cmyk-fixtures PRIVATE image_core)
  add_executable(image-cmyk-tests native_tests.cpp)
  target_link_libraries(image-cmyk-tests PRIVATE image_core)
  add_test(NAME color-managed-conversion COMMAND image-cmyk-tests)
endif()
